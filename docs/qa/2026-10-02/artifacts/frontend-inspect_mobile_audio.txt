    33	export function playNoteEvents(
    34	  events: NoteEvent[],
    35	  options: PlaybackOptions = {}
    36	): PlaybackController {
    37	  let cancelled = false;
    38	  let wake: (() => void) | null = null;
    39	  const timers: Array<ReturnType<typeof setTimeout>> = [];
    40	
    41	  const clearTimers = () => {
    42	    for (const timer of timers) clearTimeout(timer);
    43	    timers.length = 0;
    44	  };
    45	
    46	  const cancel = () => {
    47	    cancelled = true;
    48	    clearTimers();
    49	    if (wake) {
    50	      const resolve = wake;
    51	      wake = null;
    52	      resolve();
    53	    }
    54	  };
    55	
    56	  const done = (async () => {
    57	    if (events.length === 0) return;
    58	
    59	    await setAudioModeAsync({
    60	      // expo-av's playsInSilentModeIOS equivalent; recording stays off by default and the
    61	      // session ends with playback, matching the previous staysActiveInBackground: false.
    62	      playsInSilentMode: true,
    63	      interruptionMode: 'mixWithOthers',
    64	    });
    65	
    66	    const tempo = options.tempo ?? 90;
    67	    const beatDurationMs = (60 / tempo) * 1000;
    68	
    69	    const pool: AudioPlayer[] = [];
    70	    let nextVoice = 0;
    71	
    72	    // Trigger one note on the next free pool voice (round-robin), so simultaneous notes play
    73	    // concurrently instead of cutting each other off on a shared instance. Errors are
    74	    // swallowed so a mid-playback reuse can't reject the whole session.
    75	    const triggerNote = async (event: NoteEvent) => {
    76	      if (cancelled || pool.length === 0) return;
    77	      const player = pool[nextVoice % pool.length];
    78	      nextVoice += 1;
    79	
    80	      const frequency = noteToFrequency(event.note);
    81	      // Octave-folded into the usable rate window so bass/treble notes outside it
    82	      // stay distinct pitches instead of all clamping to a monotone at the boundary.
    83	      const playbackRate = computePlaybackRate(frequency);
    84	
    85	      const volume = event.velocity ?? 0.8;
    86	      // Use != null (not truthiness) so filterCutoff === 0 (a fully dark pixel) applies
    87	      // the intended 0.5x attenuation instead of falling through to full volume.
    88	      const finalVolume =
    89	        event.effects?.filterCutoff != null
    90	          ? volume * (0.5 + event.effects.filterCutoff * 0.5)
    91	          : volume;
    92	
    93	      try {
    94	        await player.seekTo(0);
    95	        // shouldCorrectPitch MUST be false: the whole pitch mechanism is varying the
    96	        // playback rate of a single beep. Pitch correction would time-stretch while
    97	        // preserving pitch, flattening every note to the sample's native pitch.
    98	        player.shouldCorrectPitch = false;
    99	        player.setPlaybackRate(playbackRate);
   100	        player.volume = Math.min(1, finalVolume);
   101	        player.play();
   102	      } catch {
   103	        // ignore — transient error (e.g. player retriggered while still playing)
   104	      }
   105	    };
   106	
   107	    try {
   108	      // Preload the sample once, then build the voice pool. createAudioPlayer is
   109	      // synchronous; preloading first means every pool player is ready to fire.
   110	      await preload(samples.beep);
   111	      for (let i = 0; i < VOICE_POOL_SIZE; i++) {
   112	        if (cancelled) return;
   113	        pool.push(createAudioPlayer(samples.beep));
   114	      }
   115	
   116	      if (cancelled) return;
   117	
   118	      // Notes carry a `start` (beats) and overlap across voices, so schedule each to fire
   119	      // at its own offset from playback start rather than playing them back-to-back.
   120	      // Playback ends once the last-finishing note's tail has elapsed.
   121	      const ordered = [...events].sort((a, b) => a.start - b.start);
   122	      const endOfLastMs =
   123	        ordered.reduce((max, e) => Math.max(max, (e.start ?? 0) + (e.duration ?? 0.5)), 0) *
   124	        beatDurationMs;
   125	
   126	      let played = 0;
   127	
   128	      await new Promise<void>((resolve) => {
   129	        if (cancelled) {
   130	          resolve();
   131	          return;
   132	        }
   133	
   134	        // cancel() clears the timers and calls wake() to resolve this promise early.
   135	        wake = resolve;
   136	
   137	        for (const event of ordered) {
   138	          const fireAt = Math.max(0, (event.start ?? 0) * beatDurationMs);
   139	          timers.push(
   140	            setTimeout(() => {
   141	              if (cancelled) return;
   142	              void triggerNote(event);
   143	              played += 1;
   144	              options.onProgress?.(played, ordered.length);
   145	            }, fireAt)
   146	          );
   147	        }
   148	
   149	        // Resolve once the last note has had time to ring out (250 ms tail). Calling an
   150	        // already-settled Promise resolve again (via cancel) is a safe no-op.
   151	        timers.push(setTimeout(resolve, endOfLastMs + 250));
   152	      });
   153	    } finally {
   154	      wake = null;
   155	      clearTimers();
   156	      // Pause + release every pooled voice, isolating per-player failures so one rejection
   157	      // doesn't leak the remaining native AudioPlayer instances.
   158	      for (const player of pool) {
   159	        try {
   160	          player.pause();
   161	        } catch {
   162	          // ignore — player may already be stopped/released
   163	        }
   164	        try {
   165	          player.remove();
   166	        } catch {
   167	          // ignore — best-effort cleanup
   168	        }
   169	      }
   170	    }
   171	  })();
   172	
   173	  return { cancel, done };
   174	}

   176	/**
   177	 * Total composition length in seconds. Notes overlap (multi-voice) and are positioned by
   178	 * `start`, so the length is the end of the last-finishing note — max(start + duration) —
   179	 * not the sum of every note's duration (which over-counts overlapping voices). Matches the
   180	 * web Tone engine's duration math.
   181	 */
   182	export function formatNoteEventsDuration(events: NoteEvent[], tempo = 90): number {
   183	  if (events.length === 0) return 0;
   184	  const beats = events.reduce(
   185	    (max, event) => Math.max(max, (event.start ?? 0) + (event.duration ?? 0.5)),
   186	    0
   187	  );
   188	  return beats * (60 / tempo);
   189	}

import { describe, expect, it } from 'vitest';
import { computePlaybackRate } from '../playbackRate';

/** Standard 12-TET frequency for a MIDI note number (A4 = 69 = 440 Hz). */
const midiToFrequency = (midi: number) => 440 * Math.pow(2, (midi - 69) / 12);

describe('computePlaybackRate', () => {
  it('maps A4 (440 Hz) to a playback rate of 1', () => {
    expect(computePlaybackRate(440)).toBeCloseTo(1, 10);
  });

  it('maps all 12 semitones of octave 2 to distinct in-range rates', () => {
    // The old flat clamp collapsed everything below 220 Hz to rate 0.5, so C2..B2 were
    // all the same pitch. Octave folding must keep each pitch class distinct.
    const rates = [];
    for (let midi = 36; midi <= 47; midi++) {
      rates.push(computePlaybackRate(midiToFrequency(midi)));
    }
    for (const rate of rates) {
      expect(rate).toBeGreaterThanOrEqual(0.5);
      expect(rate).toBeLessThanOrEqual(2.5);
    }
    expect(new Set(rates).size).toBe(12);
  });

  it('preserves pitch class by octave-folding (C2 and C3 share a rate)', () => {
    expect(computePlaybackRate(65.41)).toBeCloseTo(computePlaybackRate(130.81), 3);
  });

  it('stays within [0.5, 2.5] for MIDI 24 through 96', () => {
    for (let midi = 24; midi <= 96; midi++) {
      const rate = computePlaybackRate(midiToFrequency(midi));
      expect(rate).toBeGreaterThanOrEqual(0.5);
      expect(rate).toBeLessThanOrEqual(2.5);
    }
  });

  it('returns 1 for non-finite or non-positive input', () => {
    expect(computePlaybackRate(0)).toBe(1);
    expect(computePlaybackRate(-440)).toBe(1);
    expect(computePlaybackRate(Number.NaN)).toBe(1);
    expect(computePlaybackRate(Number.POSITIVE_INFINITY)).toBe(1);
    expect(computePlaybackRate(440, 0)).toBe(1);
    expect(computePlaybackRate(440, Number.NaN)).toBe(1);
  });

  it('returns 1 when the ratio itself underflows or overflows (no infinite folding)', () => {
    // Both inputs are finite and positive, but the ratio is 0 / Infinity — without a
    // ratio guard the folding loops would never terminate.
    expect(computePlaybackRate(Number.MIN_VALUE, Number.MAX_VALUE)).toBe(1);
    expect(computePlaybackRate(Number.MAX_VALUE, Number.MIN_VALUE)).toBe(1);
  });
});
