    18	// Explicit allowlist of client-writable composition fields. Everything else (id, userId,
    19	// createdAt, updatedAt, and any unknown keys) is derived server-side, never taken from the

    22	  'title',

    35	function pickWritableFields(input: Record<string, unknown>): Record<string, unknown> {
    36	  const out: Record<string, unknown> = {};
    37	  for (const field of WRITABLE_FIELDS) {
    38	    if (input[field] !== undefined) {
    39	      out[field] = input[field];
    40	    }
    41	  }
    42	  return out;
    43	}

   128	function encodeCursor(createdAt: string, id: string): string {
   129	  return Buffer.from(`${createdAt}|${id}`, 'utf8').toString('base64url');
   130	}

   132	/** Returns null for a malformed cursor so the route can answer 400 instead of 500. */
   133	function decodeCursor(cursor: string): { createdAt: string; id: string } | null {
   134	  let decoded: string;
   135	  try {
   136	    decoded = Buffer.from(cursor, 'base64url').toString('utf8');
   137	  } catch {
   138	    return null;
   139	  }
   140	  const sep = decoded.lastIndexOf('|');
   141	  if (sep <= 0) return null;
   142	  const createdAt = decoded.slice(0, sep);
   143	  const id = decoded.slice(sep + 1);
   144	  // created_at must round-trip as a real timestamp; id must be a UUID.
   145	  if (Number.isNaN(new Date(createdAt).getTime())) return null;
   146	  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(id)) return null;
   147	  return { createdAt, id };
   148	}

   150	export async function listCompositions(
   151	  userId: string,
   152	  options: ListCompositionsOptions = {}
   153	): Promise<ListCompositionsResult> {
   154	  // Fail closed: never list the whole table. All access is via the service-role client,
   155	  // which bypasses RLS, so this filter is the only tenant isolation.
   156	  if (!userId) {
   157	    throw new Error('listCompositions requires a userId');
   158	  }
   159	  const limit = options.limit ?? 50;
   160	  const cursor = options.cursor ? decodeCursor(options.cursor) : null;
   161	  if (options.cursor && !cursor) {
   162	    throw new Error('Invalid cursor');
   163	  }
   164	
   165	  let query = supabaseAdmin
   166	    .from('compositions')
   167	    .select(SUMMARY_SELECT)
   168	    .eq('user_id', userId)
   169	    .order('created_at', { ascending: false })
   170	    .order('id', { ascending: false });
   171	
   172	  if (cursor) {
   173	    // Keyset filter for (created_at, id) DESC: rows strictly after the cursor position.
   174	    query = query.or(
   175	      `created_at.lt.${cursor.createdAt},and(created_at.eq.${cursor.createdAt},id.lt.${cursor.id})`
   176	    );
   177	    // One extra row tells us whether another page exists.
   178	    query = query.range(0, limit);
   179	  } else {
   180	    const offset = options.offset ?? 0;
   181	    query = query.range(offset, offset + limit);
   182	  }
   183	
   184	  const { data, error } = await query;
   185	  if (error) {
   186	    throw error;
   187	  }
   188	  const rows = data as unknown as CompositionSummaryRow[];
   189	  const hasMore = rows.length > limit;
   190	  const pageRows = rows.slice(0, limit);
   191	
   192	  // With an explicit offset (no cursor) the client drives paging, so nextCursor is only
   193	  // emitted when this page provably isn't the last one. In cursor mode it is undefined on
   194	  // the final page, which is the termination signal.
   195	  const last = pageRows[pageRows.length - 1];
   196	  const nextCursor =
   197	    hasMore && last ? encodeCursor(last.created_at, last.id) : undefined;
   198	
   199	  return {
   200	    compositions: pageRows.map(mapRowToCompositionSummary),
   201	    nextCursor,
   202	  };
   203	}

   205	export async function getCompositionById(id: string): Promise<Composition | null> {
   206	  const { data, error } = await supabaseAdmin.from('compositions').select('*').eq('id', id).single();
   207	  if (error) {
   208	    if (error.code === 'PGRST116') return null;
   209	    // A malformed (non-UUID) id raises Postgres 22P02; treat it as "not found" rather than
   210	    // surfacing a 500 with internal SQL details.
   211	    if (error.code === '22P02') return null;
   212	    throw error;
   213	  }
   214	  return mapRowToComposition(data as CompositionRow);
   215	}

   242	  }

   254	  // read-then-check), closing the TOCTOU gap. NOTE: this still writes the whole data blob
   255	  // (last-write-wins across the object); a partial/JSONB merge is a follow-up.
   256	  if (!existing || existing.userId !== userId) return null;
   257	
   258	  const merged: Composition = {
   259	    ...existing,
   260	    ...pickWritableFields(updates as unknown as Record<string, unknown>),
   261	    id: existing.id,
   262	    userId: existing.userId,
   263	    createdAt: existing.createdAt,
   264	    updatedAt: new Date(),
   265	  };
   266	
   267	  const { data, error } = await supabaseAdmin
   268	    .from('compositions')
   269	    .update({
   270	      name: merged.title,
   271	      data: merged,
   272	      updated_at: merged.updatedAt.toISOString(),
   273	    })
   274	    .eq('id', id)
   275	    .eq('user_id', userId)
   276	    .select()
   277	    .single();
   278	
   279	  if (error) {
   280	    if (error.code === 'PGRST116' || error.code === '22P02') return null;
   281	    throw error;
   282	  }
   283	
   284	  return mapRowToComposition(data as CompositionRow);
   285	}
   286	
   287	export async function deleteComposition(id: string, userId: string): Promise<boolean> {
   288	  const { data, error } = await supabaseAdmin
   289	    .from('compositions')
   290	    .delete()
   291	    .eq('id', id)
   292	    .eq('user_id', userId)
   293	    .select('id');
   294	  if (error) {
   295	    if (error.code === '22P02') return false;
   296	    throw error;
   297	  }
   298	  // Returns the deleted rows; empty means nothing matched (missing or not owned).
   299	  return Array.isArray(data) && data.length > 0;
   300	}

describe('POST /compositions (validation, #88)', () => {
  it('creates on a valid body', async () => {
    mocked.createComposition.mockResolvedValue(composition(TEST_USER));
    const res = await app.inject({ method: 'POST', url: '/compositions', payload: validCreateBody });
    expect(res.statusCode).toBe(201);
    expect(mocked.createComposition).toHaveBeenCalledWith(TEST_USER, expect.objectContaining({ title: 'My Comp' }));
  });

  it('rejects a missing required field with 400', async () => {
    const { title, ...noTitle } = validCreateBody;
    void title;
    const res = await app.inject({ method: 'POST', url: '/compositions', payload: noTitle });
    expect(res.statusCode).toBe(400);
    expect(mocked.createComposition).not.toHaveBeenCalled();
  });

  it('strips unknown/mass-assignment keys so they never reach the service', async () => {
    // With additionalProperties:false, Fastify's default AJV removes unknown keys rather
    // than 400-ing; either way the forged userId/id must not reach createComposition.
    mocked.createComposition.mockResolvedValue(composition(TEST_USER));
    const res = await app.inject({
      method: 'POST',
      url: '/compositions',
      payload: { ...validCreateBody, userId: OTHER_USER, id: 'forged' },
    });
    expect(res.statusCode).toBe(201);
    const body = mocked.createComposition.mock.calls[0][1] as Record<string, unknown>;
    expect(body).not.toHaveProperty('userId');
    expect(body).not.toHaveProperty('id');
  });

  it('rejects a wrong-typed field with 400', async () => {
    const res = await app.inject({
      method: 'POST',
      url: '/compositions',
      payload: { ...validCreateBody, title: { not: 'a string' } },
    });
    expect(res.statusCode).toBe(400);
  });

  it('rejects a note event missing "note" with 400', async () => {
    const res = await app.inject({
      method: 'POST',
      url: '/compositions',
      payload: { ...validCreateBody, noteEvents: [{ start: 0, duration: 0.5, velocity: 0.8 }] },
    });
    expect(res.statusCode).toBe(400);
    expect(mocked.createComposition).not.toHaveBeenCalled();
  });

  it('keeps unknown note-event keys (forward compat) while enforcing the core fields', async () => {
    mocked.createComposition.mockResolvedValue(composition(TEST_USER));
    const res = await app.inject({
      method: 'POST',
      url: '/compositions',
      payload: {
        ...validCreateBody,
        noteEvents: [{ note: 'C4', start: 0, duration: 0.5, velocity: 0.8, futureField: 'x' }],
      },
    });
    expect(res.statusCode).toBe(201);
    const body = mocked.createComposition.mock.calls[0][1] as { noteEvents: Record<string, unknown>[] };
    expect(body.noteEvents[0]).toHaveProperty('futureField', 'x');
  });

import type { FastifyInstance } from 'fastify';
import type { HealthCheckResponse } from '@toposonics/types';
import { config } from '../config';
import { requireAuth } from '../auth';

export async function healthRoutes(fastify: FastifyInstance) {
  /**
   * GET /health
   * Basic health check endpoint
   */
  fastify.get<{ Reply: HealthCheckResponse }>('/health', async (_request, reply) => {
    const uptime = process.uptime();

    const response: HealthCheckResponse = {
      status: 'ok',
      timestamp: new Date().toISOString(),
      version: config.apiVersion,
      uptime: Math.floor(uptime),
    };

    return reply.send(response);
  });

  /**
   * GET /health/detailed
   * More detailed health information. Gated behind auth so environment/uptime/memory
   * internals aren't exposed to anonymous clients (fingerprinting / memory-probe surface).
   */
  fastify.get('/health/detailed', { preHandler: requireAuth }, async (_request, reply) => {
    const uptime = process.uptime();
    const memoryUsage = process.memoryUsage();

    return reply.send({
      status: 'ok',
      timestamp: new Date().toISOString(),
      version: config.apiVersion,
      uptime: Math.floor(uptime),
      environment: config.isProduction ? 'production' : 'development',
      memory: {
        heapUsed: Math.round(memoryUsage.heapUsed / 1024 / 1024) + ' MB',
        heapTotal: Math.round(memoryUsage.heapTotal / 1024 / 1024) + ' MB',
        rss: Math.round(memoryUsage.rss / 1024 / 1024) + ' MB',
      },
    });
  });
}
