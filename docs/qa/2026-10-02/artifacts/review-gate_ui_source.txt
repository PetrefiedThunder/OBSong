    10	const STORAGE_KEY_PREFIX = 'supabase.auth.';
    11	const CHUNK_SUFFIX = '.chunk.';
    12	const CHUNK_COUNT_SUFFIX = '.chunks';
    13	
    14	// expo-secure-store 12.x (SDK 50) caps values at 2048 bytes, and some Android
    15	// keystore implementations hard-fail above it. A Supabase session blob (access JWT +
    16	// refresh token + serialized user) is routinely 3-5 KB, so values that exceed the limit
    17	// are split across numbered SecureStore keys instead of being silently dropped/truncated.
    18	const SECURE_STORE_VALUE_LIMIT = 2048;
    19	
    20	interface ChunkedRef {
    21	  chunks: number;
    22	}
    23	
    24	async function removeChunks(prefixedKey: string): Promise<void> {
    25	  const countRaw = await AsyncStorage.getItem(prefixedKey + CHUNK_COUNT_SUFFIX);
    26	  if (countRaw == null) return;
    27	
    28	  const count = Number.parseInt(countRaw, 10);
    29	  for (let i = 0; i < count; i++) {
    30	    try {
    31	      await SecureStore.deleteItemAsync(prefixedKey + CHUNK_SUFFIX + i);
    32	    } catch {
    33	      // best-effort cleanup
    34	    }
    35	  }
    36	  await AsyncStorage.removeItem(prefixedKey + CHUNK_COUNT_SUFFIX);
    37	}

    39	export const secureStorage: SupportedStorage = {
    40	  /**
    41	   * Store a value securely in the device's encrypted storage
    42	   */
    43	  async setItem(key: string, value: string): Promise<void> {
    44	    const prefixedKey = STORAGE_KEY_PREFIX + key;
    45	    try {
    46	      if (value.length <= SECURE_STORE_VALUE_LIMIT) {
    47	        // Fits a single key: store directly and clear any stale chunks from a
    48	        // previous oversized value so getItem doesn't reassemble them.
    49	        await removeChunks(prefixedKey);
    50	        await SecureStore.setItemAsync(prefixedKey, value);
    51	        return;
    52	      }
    53	
    54	      const chunkCount = Math.ceil(value.length / SECURE_STORE_VALUE_LIMIT);
    55	      for (let i = 0; i < chunkCount; i++) {
    56	        await SecureStore.setItemAsync(
    57	          prefixedKey + CHUNK_SUFFIX + i,
    58	          value.slice(i * SECURE_STORE_VALUE_LIMIT, (i + 1) * SECURE_STORE_VALUE_LIMIT)
    59	        );
    60	      }
    61	
    62	      // The ref goes through AsyncStorage because it is JSON (a SecureStore value is a
    63	      // plain string, and parsing "null"/"{" is ambiguous with raw stored values).
    64	      await AsyncStorage.setItem(
    65	        prefixedKey + CHUNK_COUNT_SUFFIX,
    66	        JSON.stringify({ chunks: chunkCount } satisfies ChunkedRef)
    67	      );
    68	      // Clear the single-key slot so an old small value can't shadow the chunks.
    69	      await SecureStore.deleteItemAsync(prefixedKey);
    70	    } catch (error) {
    71	      console.error('SecureStore setItem error:', error);
    72	      throw error;
    73	    }
    74	  },
    75	
    76	  /**
    77	   * Retrieve a value from secure storage
    78	   */
    79	  async getItem(key: string): Promise<string | null> {
    80	    const prefixedKey = STORAGE_KEY_PREFIX + key;
    81	    try {
    82	      const refRaw = await AsyncStorage.getItem(prefixedKey + CHUNK_COUNT_SUFFIX);
    83	      if (refRaw == null) {
    84	        return await SecureStore.getItemAsync(prefixedKey);
    85	      }
    86	
    87	      const ref = JSON.parse(refRaw) as ChunkedRef;
    88	      const parts: string[] = [];
    89	      for (let i = 0; i < ref.chunks; i++) {
    90	        const part = await SecureStore.getItemAsync(prefixedKey + CHUNK_SUFFIX + i);
    91	        if (part == null) {
    92	          // A chunk vanished (keystore reset, partial wipe) — the session is
    93	          // unrecoverable; report null rather than a corrupt partial blob.
    94	          return null;
    95	        }
    96	        parts.push(part);
    97	      }
    98	      return parts.join('');
    99	    } catch (error) {
   100	      console.error('SecureStore getItem error:', error);
   101	      return null;
   102	    }
   103	  },
   104	
   105	  /**
   106	   * Remove a value from secure storage
   107	   */
   108	  async removeItem(key: string): Promise<void> {
   109	    const prefixedKey = STORAGE_KEY_PREFIX + key;
   110	    try {
   111	      await removeChunks(prefixedKey);
   112	      await SecureStore.deleteItemAsync(prefixedKey);
   113	    } catch (error) {
   114	      console.error('SecureStore removeItem error:', error);
   115	      throw error;
   116	    }
   117	  },

    36	 * Extract pixel data from an image element
    37	 */
    38	export function extractPixelData(img: HTMLImageElement): {
    39	  pixels: Uint8ClampedArray;
    40	  width: number;
    41	  height: number;
    42	} {
    43	  const canvas = document.createElement('canvas');
    44	  const ctx = canvas.getContext('2d');
    45	
    46	  if (!ctx) {
    47	    throw new Error('Failed to get canvas context');
    48	  }
    49	
    50	  // Resize for performance if image is very large
    51	  const maxDimension = 1200;
    52	  let width = img.width;
    53	  let height = img.height;
    54	
    55	  if (width > maxDimension || height > maxDimension) {
    56	    if (width > height) {
    57	      height = (height / width) * maxDimension;
    58	      width = maxDimension;
    59	    } else {
    60	      width = (width / height) * maxDimension;
    61	      height = maxDimension;
    62	    }
    63	  }
    64	
    65	  canvas.width = width;
    66	  canvas.height = height;
    67	
    68	  ctx.drawImage(img, 0, 0, width, height);
    69	
    70	  const imageData = ctx.getImageData(0, 0, width, height);
    71	
    72	  return {
    73	    pixels: imageData.data,
    74	    width,
    75	    height,
    76	  };
