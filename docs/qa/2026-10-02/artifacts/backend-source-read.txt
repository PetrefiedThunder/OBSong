     9	// Short-TTL cache of token -> resolved user so repeated requests from the same client
    10	// don't each make a blocking network call to Supabase Auth (which would double per-request
    11	// latency and couple every authed endpoint to Supabase availability / rate limits). A
    12	// 30s TTL bounds token-revocation lag; the size cap prevents unbounded growth.
    13	const TOKEN_CACHE_TTL_MS = 30_000;
    14	const TOKEN_CACHE_MAX = 5_000;
    15	const tokenCache = new Map<string, { user: User | null; expiresAt: number }>();
    16	
    17	export async function getUserFromToken(token: string): Promise<User | null> {
    18	  const now = Date.now();
    19	  const cached = tokenCache.get(token);
    20	  if (cached && cached.expiresAt > now) {
    21	    return cached.user;
    22	  }
    23	
    24	  const { data, error } = await supabaseAdmin.auth.getUser(token);
    25	
    26	  let user: User | null = null;
    27	  if (!error && data.user) {
    28	    const supabaseUser = data.user;
    29	    user = {
    30	      id: supabaseUser.id,
    31	      email: supabaseUser.email || '',
    32	      displayName: (supabaseUser.user_metadata as Record<string, unknown>)?.full_name as
    33	        | string
    34	        | undefined,
    35	      createdAt: new Date(supabaseUser.created_at),
    36	      lastLoginAt: supabaseUser.last_sign_in_at
    37	        ? new Date(supabaseUser.last_sign_in_at)
    38	        : undefined,
    39	    };
    40	  }
    41	
    42	  // Bound the cache size (drop the oldest entry — Map preserves insertion order).
    43	  if (tokenCache.size >= TOKEN_CACHE_MAX) {
    44	    const oldest = tokenCache.keys().next().value;
    45	    if (oldest !== undefined) tokenCache.delete(oldest);
    46	  }
    47	  tokenCache.set(token, { user, expiresAt: now + TOKEN_CACHE_TTL_MS });
    48	
    49	  return user;
    50	}

    55	export async function requireAuth(request: FastifyRequest, reply: FastifyReply) {
    56	  const authHeader = request.headers.authorization;
    57	
    58	  if (!authHeader) {
    59	    return reply.status(401).send({
    60	      success: false,
    61	      error: {
    62	        code: 'UNAUTHORIZED',
    63	        message: 'Missing authorization header',
    64	      },
    65	    });
    66	  }
    67	
    68	  const parts = authHeader.split(' ');
    69	  if (parts.length !== 2 || parts[0] !== 'Bearer') {
    70	    return reply.status(401).send({
    71	      success: false,
    72	      error: {
    73	        code: 'INVALID_AUTH_HEADER',
    74	        message: 'Authorization header must be "Bearer <token>"',
    75	      },
    76	    });
    77	  }
    78	
    79	  const token = parts[1];
    80	  const user = await getUserFromToken(token);
    81	
    82	  if (!user) {
    83	    return reply.status(401).send({
    84	      success: false,
    85	      error: {
    86	        code: 'INVALID_TOKEN',
    87	        message: 'Invalid or expired Supabase token',
    88	      },
    89	    });
    90	  }
    91	
    92	  request.userId = user.id;
    93	  request.user = user;
    94	}

     7	 * (don't trust client-supplied forwarding headers). Accepts `true`/`false`, a hop count

    10	// Matches Fastify's `trustProxy` option type (fastify v5.12 dropped numeric hop counts).
    11	function parseTrustProxy(value: string | undefined): boolean | string {
    12	  if (!value) return false;
    13	  if (value === 'true') return true;
    14	  if (value === 'false') return false;
    15	  const asNumber = Number(value);
    16	  if (Number.isInteger(asNumber) && asNumber >= 0) return asNumber > 0;
    17	  return value;
    18	}

   109	export interface ListCompositionsOptions {
   110	  /** Page size (rows to return). */
   111	  limit?: number;
   112	  /** Rows to skip before the page (offset pagination; ignored when cursor is set). */
   113	  offset?: number;
   114	  /**
   115	   * Opaque cursor from a previous page's `nextCursor` (base64url of `<isoCreatedAt>|<id>`).
   116	   * When present, results are keyset-paginated by the stable (created_at DESC, id DESC)
   117	   * ordering and offset is ignored.
   118	   */
   119	  cursor?: string;

   122	export interface ListCompositionsResult {
   123	  compositions: CompositionSummary[];
   124	  /** Cursor to pass for the next page; undefined on the last page. */
   125	  nextCursor?: string;
   126	}

   128	function encodeCursor(createdAt: string, id: string): string {
   129	  return Buffer.from(`${createdAt}|${id}`, 'utf8').toString('base64url');
   130	}
   131	
   132	/** Returns null for a malformed cursor so the route can answer 400 instead of 500. */
   133	function decodeCursor(cursor: string): { createdAt: string; id: string } | null {
   134	  let decoded: string;
   135	  try {
   136	    decoded = Buffer.from(cursor, 'base64url').toString('utf8');
   137	  } catch {
   138	    return null;
   139	  }
   140	  const sep = decoded.lastIndexOf('|');
   141	  if (sep <= 0) return null;
   142	  const createdAt = decoded.slice(0, sep);
   143	  const id = decoded.slice(sep + 1);
   144	  // created_at must round-trip as a real timestamp; id must be a UUID.
   145	  if (Number.isNaN(new Date(createdAt).getTime())) return null;
   146	  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(id)) return null;
   147	  return { createdAt, id };
   148	}

   150	export async function listCompositions(
   151	  userId: string,
   152	  options: ListCompositionsOptions = {}
   153	): Promise<ListCompositionsResult> {
   154	  // Fail closed: never list the whole table. All access is via the service-role client,
   155	  // which bypasses RLS, so this filter is the only tenant isolation.
   156	  if (!userId) {
   157	    throw new Error('listCompositions requires a userId');
   158	  }
   159	  const limit = options.limit ?? 50;
   160	  const cursor = options.cursor ? decodeCursor(options.cursor) : null;
   161	  if (options.cursor && !cursor) {
   162	    throw new Error('Invalid cursor');
   163	  }
   164	
   165	  let query = supabaseAdmin
   166	    .from('compositions')
   167	    .select(SUMMARY_SELECT)
   168	    .eq('user_id', userId)
   169	    .order('created_at', { ascending: false })
   170	    .order('id', { ascending: false });
   171	
   172	  if (cursor) {
   173	    // Keyset filter for (created_at, id) DESC: rows strictly after the cursor position.
   174	    query = query.or(
   175	      `created_at.lt.${cursor.createdAt},and(created_at.eq.${cursor.createdAt},id.lt.${cursor.id})`
   176	    );
   177	    // One extra row tells us whether another page exists.
   178	    query = query.range(0, limit);
   179	  } else {
   180	    const offset = options.offset ?? 0;
   181	    query = query.range(offset, offset + limit);
   182	  }
   183	
   184	  const { data, error } = await query;
   185	  if (error) {
   186	    throw error;
   187	  }
   188	  const rows = data as unknown as CompositionSummaryRow[];
   189	  const hasMore = rows.length > limit;
   190	  const pageRows = rows.slice(0, limit);
   191	
   192	  // With an explicit offset (no cursor) the client drives paging, so nextCursor is only
   193	  // emitted when this page provably isn't the last one. In cursor mode it is undefined on
   194	  // the final page, which is the termination signal.
   195	  const last = pageRows[pageRows.length - 1];
   196	  const nextCursor =
   197	    hasMore && last ? encodeCursor(last.created_at, last.id) : undefined;
   198	
   199	  return {
   200	    compositions: pageRows.map(mapRowToCompositionSummary),
   201	    nextCursor,
   202	  };
   203	}

   205	export async function getCompositionById(id: string): Promise<Composition | null> {
   206	  const { data, error } = await supabaseAdmin.from('compositions').select('*').eq('id', id).single();
   207	  if (error) {
   208	    if (error.code === 'PGRST116') return null;
   209	    // A malformed (non-UUID) id raises Postgres 22P02; treat it as "not found" rather than
   210	    // surfacing a 500 with internal SQL details.
   211	    if (error.code === '22P02') return null;
   212	    throw error;
   213	  }
   214	  return mapRowToComposition(data as CompositionRow);
   215	}

   217	export async function createComposition(
   218	  userId: string,
   219	  payload: Omit<CreateCompositionDTO, 'id' | 'createdAt' | 'updatedAt' | 'userId'>
   220	): Promise<Composition> {
   221	  const now = new Date();
   222	  const compositionPayload: Composition = {
   223	    ...pickWritableFields(payload as unknown as Record<string, unknown>),
   224	    id: '',
   225	    userId,
   226	    createdAt: now,
   227	    updatedAt: now,
   228	  } as Composition;
   229	
   230	  const { data, error } = await supabaseAdmin
   231	    .from('compositions')
   232	    .insert({
   233	      user_id: userId,
   234	      name: payload.title,
   235	      data: compositionPayload,
   236	    })
   237	    .select()
   238	    .single();
   239	
   240	  if (error) {
   241	    throw error;
   242	  }
   243	
   244	  return mapRowToComposition(data as CompositionRow);
   245	}

   247	export async function updateComposition(
   248	  id: string,
   249	  userId: string,
   250	  updates: UpdateCompositionDTO
   251	): Promise<Composition | null> {
   252	  const existing = await getCompositionById(id);
   253	  // Scope by owner so the mutating query is defended in depth (not only by the route's
   254	  // read-then-check), closing the TOCTOU gap. NOTE: this still writes the whole data blob
   255	  // (last-write-wins across the object); a partial/JSONB merge is a follow-up.
   256	  if (!existing || existing.userId !== userId) return null;
   257	
   258	  const merged: Composition = {
   259	    ...existing,
   260	    ...pickWritableFields(updates as unknown as Record<string, unknown>),
   261	    id: existing.id,
   262	    userId: existing.userId,
   263	    createdAt: existing.createdAt,
   264	    updatedAt: new Date(),
   265	  };
   266	
   267	  const { data, error } = await supabaseAdmin
   268	    .from('compositions')
   269	    .update({
   270	      name: merged.title,
   271	      data: merged,
   272	      updated_at: merged.updatedAt.toISOString(),
   273	    })
   274	    .eq('id', id)
   275	    .eq('user_id', userId)
   276	    .select()
   277	    .single();
   278	
   279	  if (error) {
   280	    if (error.code === 'PGRST116' || error.code === '22P02') return null;
   281	    throw error;
   282	  }
   283	
   284	  return mapRowToComposition(data as CompositionRow);
   285	}
   286	
   287	export async function deleteComposition(id: string, userId: string): Promise<boolean> {
   288	  const { data, error } = await supabaseAdmin
   289	    .from('compositions')
   290	    .delete()
   291	    .eq('id', id)
   292	    .eq('user_id', userId)
   293	    .select('id');
   294	  if (error) {
   295	    if (error.code === '22P02') return false;
   296	    throw error;
   297	  }
   298	  // Returns the deleted rows; empty means nothing matched (missing or not owned).
   299	  return Array.isArray(data) && data.length > 0;
   300	}

    22	interface Builder {
    23	  eqCalls: Array<[string, unknown]>;
    24	  selectCalls: string[];
    25	  rangeCalls: Array<[number, number]>;
    26	  orderCalls: Array<[string, unknown]>;
    27	  orCalls: string[];
    28	  then: (resolve: (v: unknown) => unknown, reject?: (e: unknown) => unknown) => Promise<unknown>;
    29	  [k: string]: unknown;
    30	}
    31	
    32	function makeBuilder(result: unknown): Builder {
    33	  const eqCalls: Array<[string, unknown]> = [];
    34	  const selectCalls: string[] = [];
    35	  const rangeCalls: Array<[number, number]> = [];
    36	  const orderCalls: Array<[string, unknown]> = [];
    37	  const orCalls: string[] = [];
    38	  const builder = { eqCalls, selectCalls, rangeCalls, orderCalls, orCalls } as Builder;
    39	  for (const m of ['select', 'insert', 'update', 'delete', 'order', 'single', 'eq', 'range', 'or']) {
    40	    builder[m] = vi.fn((...args: unknown[]) => {
    41	      if (m === 'eq') eqCalls.push([args[0] as string, args[1]]);
    42	      if (m === 'select') selectCalls.push(args[0] as string);
    43	      if (m === 'range') rangeCalls.push([args[0] as number, args[1] as number]);
    44	      if (m === 'order') orderCalls.push([args[0] as string, args[1]]);
    45	      if (m === 'or') orCalls.push(args[0] as string);
    46	      return builder;
    47	    });
    48	  }
    49	  builder.then = (resolve, reject) => Promise.resolve(result).then(resolve, reject);
    50	  return builder;
    51	}

     1	# @toposonics/api
     2	
     3	Fastify API for TopoSonics with Supabase-backed authentication and Postgres persistence.
     4	
     5	## What It Does
     6	
     7	- Exposes health and status endpoints
     8	- Validates Supabase access tokens
     9	- Stores private user compositions
    10	- Enforces ownership for composition reads/writes

    34	## Endpoints
    35	
    36	### Health
    37	
    38	- `GET /health`
    39	- `GET /health/detailed`
    40	
    41	### Compositions
    42	
    43	All composition routes require `Authorization: Bearer <accessToken>`.
    44	
    45	- `GET /compositions` lists the authenticated user's private compositions
    46	- `GET /compositions/:id` returns one owned composition
    47	- `POST /compositions` creates a composition for the authenticated user
    48	- `PUT /compositions/:id` updates an owned composition
    49	- `DELETE /compositions/:id` deletes an owned composition
    50	
    51	Example payload:
    52	
    53	```json
    54	{
    55	  "title": "My Composition",
    56	  "description": "Generated from mountain image",
    57	  "noteEvents": [],
    58	  "mappingMode": "LINEAR_LANDSCAPE",
    59	  "key": "C",
    60	  "scale": "C_MAJOR",
    61	  "presetId": "sine-soft",
    62	  "tempo": 90
    63	}
    64	```
    65	
    66	## Product Contract
    67	
    68	- There is no public gallery route in this API pass.
    69	- Composition storage is private-only in v1.
    70	- Public/demo discovery is handled by static client-side demos, not anonymous composition API access.
