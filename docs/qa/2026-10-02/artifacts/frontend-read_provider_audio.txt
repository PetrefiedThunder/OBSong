    69	    import('tone/build/esm/event/Part.js'),
    70	    import('tone/build/esm/core/Global.js'),
    71	  ]).then(([polySynth, synth, reverb, filter, panner, feedbackDelay, part, global]) => {
    72	    const context = global.getContext();
    73	
    74	    return {
    75	      PolySynth: polySynth.PolySynth as ToneModule['PolySynth'],
    76	      Synth: synth.Synth as ToneModule['Synth'],
    77	      Reverb: reverb.Reverb as ToneModule['Reverb'],
    78	      Filter: filter.Filter as ToneModule['Filter'],
    79	      Panner: panner.Panner as ToneModule['Panner'],
    80	      FeedbackDelay: feedbackDelay.FeedbackDelay as ToneModule['FeedbackDelay'],
    81	      Part: part.Part as ToneModule['Part'],
    82	      Destination: context.destination,
    83	      Transport: context.transport as ToneTransport,
    84	      start: global.start,
    85	    };
    86	  });
    87	  return tonePromise;

    91	  noteEvents: NoteEvent[];

    96	interface VoiceSynth {
    97	  synth: ToneSynth;
    98	  reverb: ToneReverb;
    99	  filter: ToneFilter;
   100	  panner: TonePanner;
   101	  delay: ToneFeedbackDelay | null;
   102	}
   103	
   104	// Tone's FeedbackDelay defaults to a fully wet mix, which would REPLACE the dry signal
   105	// with delayed repeats (shifting every note by the delay time). A partial wet layers
   106	// echoes over the original instead; presets only declare time/feedback, so the mix is
   107	// fixed here.
   108	const DELAY_WET = 0.35;
   109	
   110	// NoteEvent.effects.filterCutoff is normalized 0-1 (brightness-driven). Map it onto an
   111	// exponential frequency curve (~200 Hz at 0, ~8 kHz at 1) so equal cutoff steps sound
   112	// like equal changes in brightness.
   113	function filterCutoffToFrequency(cutoff: number): number {
   114	  return 200 * Math.pow(40, cutoff);
   115	}

   117	export function useToneEngine({ noteEvents, tempo, preset }: UseToneEngineOptions) {
   118	  const [isPlaying, setIsPlaying] = useState(false);
   119	  const [isLoading, setIsLoading] = useState(false);
   120	  const [currentTime, setCurrentTime] = useState(0);
   121	
   122	  // Legacy single-synth refs (for backward compatibility)
   123	  const toneRef = useRef<ToneModule | null>(null);
   124	  const synthRef = useRef<ToneSynth | null>(null);
   125	  const reverbRef = useRef<ToneReverb | null>(null);
   126	  const filterRef = useRef<ToneFilter | null>(null);
   127	  const pannerRef = useRef<TonePanner | null>(null);
   128	  const delayRef = useRef<ToneFeedbackDelay | null>(null);
   129	
   130	  // Multi-voice synth refs
   131	  const voiceSynthsRef = useRef<Map<VoiceType, VoiceSynth>>(new Map());
   132	
   133	  const partRef = useRef<TonePart | null>(null);
   134	  const stopSchedulerIdRef = useRef<number | null>(null);
   135	  const graphSignatureRef = useRef<string | null>(null);
   136	
   137	  // Detect if we're in multi-voice mode
   138	  const isMultiVoice = getAudioTopology(noteEvents) === 'multi';
   139	  const graphSignature = getAudioGraphSignature(noteEvents, preset);
   140	
   141	  const disposePart = useCallback(() => {
   142	    if (partRef.current) {
   143	      partRef.current.dispose();
   144	      partRef.current = null;
   145	    }
   146	  }, []);
   147	
   148	  const disposeAudioGraph = useCallback(() => {
   149	    disposePart();
   150	
   151	    if (synthRef.current) {
   152	      synthRef.current.dispose();
   153	      synthRef.current = null;
   154	    }
   155	    if (reverbRef.current) {
   156	      reverbRef.current.dispose();
   157	      reverbRef.current = null;
   158	    }
   159	    if (filterRef.current) {
   160	      filterRef.current.dispose();
   161	      filterRef.current = null;
   162	    }
   163	    if (pannerRef.current) {
   164	      pannerRef.current.dispose();
   165	      pannerRef.current = null;
   166	    }
   167	    if (delayRef.current) {
   168	      delayRef.current.dispose();
   169	      delayRef.current = null;
   170	    }
   171	
   172	    for (const voiceSynth of voiceSynthsRef.current.values()) {
   173	      voiceSynth.synth.dispose();
   174	      voiceSynth.reverb.dispose();
   175	      voiceSynth.filter.dispose();
   176	      voiceSynth.panner.dispose();
   177	      voiceSynth.delay?.dispose();
   178	    }
   179	    voiceSynthsRef.current.clear();
   180	
   181	    graphSignatureRef.current = null;
   182	  }, [disposePart]);
   183	
   184	  const stop = useCallback(() => {
   185	    const tone = toneRef.current;
   186	
   187	    if (tone && stopSchedulerIdRef.current !== null) {
   188	      tone.Transport.clear(stopSchedulerIdRef.current);
   189	      stopSchedulerIdRef.current = null;
   190	    }
   191	
   192	    if (partRef.current) {
   193	      partRef.current.stop();
   194	    }
   195	
   196	    if (tone) {
   197	      tone.Transport.stop();
   198	      tone.Transport.position = 0;
   199	    }
   200	
   201	    setIsPlaying(false);
   202	    setCurrentTime(0);
   203	  }, []);
   204	
   205	  const createMultiVoiceGraph = useCallback(async () => {
   206	    const voices: VoiceType[] = ['bass', 'melody', 'pad'];
   207	
   208	    for (const voice of voices) {
   209	      let synthConfig;
   210	      let reverbConfig;
   211	
   212	      switch (voice) {
   213	        case 'bass':
   214	          synthConfig = {
   215	            oscillator: { type: 'triangle' as const },
   216	            envelope: { attack: 0.05, decay: 0.1, sustain: 0.8, release: 1.0 },
   217	          };
   218	          reverbConfig = { decay: 1.5, wet: 0.15 };
   219	          break;
   220	
   221	        case 'melody':
   222	          synthConfig = {
   223	            oscillator: { type: preset.oscillatorType },
   224	            envelope: preset.synthesis?.envelope || {
   225	              attack: 0.01,
   226	              decay: 0.1,
   227	              sustain: 0.3,
   228	              release: 0.5,
   229	            },
   230	          };
   231	          reverbConfig = { decay: 2.5, wet: 0.3 };
   232	          break;
   233	
   234	        case 'pad':
   235	          synthConfig = {
   236	            oscillator: { type: 'sine' as const },
   237	            envelope: { attack: 0.5, decay: 0.2, sustain: 0.7, release: 2.0 },
   238	          };
   239	          reverbConfig = { decay: 4.0, wet: 0.5 };
   240	          break;
   241	
   242	        default:
   243	          synthConfig = {
   244	            oscillator: { type: preset.oscillatorType },
   245	            envelope: preset.synthesis?.envelope || {
   246	              attack: 0.1,
   247	              decay: 0.2,
   248	              sustain: 0.6,
   249	              release: 0.8,
   250	            },
   251	          };
   252	          reverbConfig = { decay: 2, wet: 0.2 };
   253	      }
   254	
   255	      const tone = toneRef.current ?? (await loadTone());
   256	      toneRef.current = tone;
   257	
   258	      const synth = new tone.PolySynth(tone.Synth, synthConfig) as ToneSynth;
   259	      const reverb = new tone.Reverb(reverbConfig) as ToneReverb;
   260	      await reverb.generate();
   261	
   262	      // Every voice gets its own Filter + Panner so per-note filterCutoff and pan apply
   263	      // to bass and pad, not just melody. Known limitation: simultaneous chord notes on
   264	      // one voice share the Panner, so intra-chord stereo spread collapses to the
   265	      // last-set value — pan over time still works.
   266	      const filter = new tone.Filter({
   267	        type: 'lowpass',
   268	        frequency: 3000,
   269	        Q: 1,
   270	      }) as ToneFilter;
   271	      const panner = new tone.Panner(0) as TonePanner;
   272	
   273	      // Presets may declare a delay effect; give each voice its own FeedbackDelay so
   274	      // multi-voice compositions render it too (before reverb, sharing its tail).
   275	      const delayConfig = preset.synthesis?.effects?.delay;
   276	      const delay = delayConfig
   277	        ? (new tone.FeedbackDelay({
   278	            delayTime: delayConfig.time,
   279	            feedback: delayConfig.feedback,
   280	            wet: DELAY_WET,
   281	          }) as ToneFeedbackDelay)
   282	        : null;
   283	
   284	      if (delay) {
   285	        synth.chain(filter, panner, delay, reverb, tone.Destination);
   286	      } else {
   287	        synth.chain(filter, panner, reverb, tone.Destination);
   288	      }
   289	
   290	      voiceSynthsRef.current.set(voice, { synth, reverb, filter, panner, delay });
   291	    }
   292	  }, [preset]);
   293	
   294	  const createSingleVoiceGraph = useCallback(async () => {
   295	    const tone = toneRef.current ?? (await loadTone());
   296	    toneRef.current = tone;
   297	
   298	    const synth = new tone.PolySynth(tone.Synth, {
   299	      oscillator: {
   300	        type: preset.oscillatorType,
   301	      },
   302	      envelope: preset.synthesis?.envelope || {
   303	        attack: 0.1,
   304	        decay: 0.2,
   305	        sustain: 0.6,
   306	        release: 0.8,
   307	      },
   308	    }) as ToneSynth;
   309	
   310	    const reverb = new tone.Reverb({
   311	      decay: preset.synthesis?.effects?.reverb?.decay || 2,
   312	      wet: 0,
   313	    }) as ToneReverb;
   314	
   315	    const filter = new tone.Filter({
   316	      type: preset.synthesis?.filter?.type || 'lowpass',
   317	      frequency: preset.synthesis?.filter?.frequency || 2000,
   318	      Q: preset.synthesis?.filter?.resonance || 1,
   319	    }) as ToneFilter;
   320	
   321	    const panner = new tone.Panner(0) as TonePanner;
   322	
   323	    // Presets may declare a delay effect; insert a FeedbackDelay into the chain so it
   324	    // actually sounds (before reverb, so echoes share the preset's reverb tail). Partial
   325	    // wet — the default fully-wet mix would replace the dry signal with shifted repeats.
   326	    const delayConfig = preset.synthesis?.effects?.delay;
   327	    const delay = delayConfig
   328	      ? (new tone.FeedbackDelay({
   329	          delayTime: delayConfig.time,
   330	          feedback: delayConfig.feedback,
   331	          wet: DELAY_WET,
   332	        }) as ToneFeedbackDelay)
   333	      : null;
   334	
   335	    if (delay) {
   336	      synth.chain(filter, panner, delay, reverb, tone.Destination);
   337	    } else {
   338	      synth.chain(filter, panner, reverb, tone.Destination);
   339	    }
   340	
   341	    synthRef.current = synth;
   342	    reverbRef.current = reverb;
   343	    filterRef.current = filter;
   344	    pannerRef.current = panner;
   345	    delayRef.current = delay;
   346	  }, [preset]);
   347	
   348	  const ensureAudioGraph = useCallback(async () => {
   349	    const tone = await loadTone();
   350	    toneRef.current = tone;
   351	
   352	    await tone.start();
   353	
   354	    if (graphSignatureRef.current === graphSignature) {
   355	      return;
   356	    }
   357	
   358	    setIsLoading(true);
   359	    stop();
   360	    disposeAudioGraph();
   361	
   362	    try {
   363	      if (isMultiVoice) {
   364	        await createMultiVoiceGraph();
   365	      } else {
   366	        await createSingleVoiceGraph();
   367	      }
   368	
   369	      graphSignatureRef.current = graphSignature;
   370	    } catch (error) {
   371	      disposeAudioGraph();
   372	      throw error;
   373	    } finally {
   374	      setIsLoading(false);
   375	    }
   376	  }, [
   377	    createMultiVoiceGraph,
   378	    createSingleVoiceGraph,
   379	    disposeAudioGraph,
   380	    graphSignature,
   381	    isMultiVoice,
   382	    stop,
   383	  ]);
   384	
   385	  const buildPart = useCallback(() => {
   386	    disposePart();
   387	
   388	    if (!noteEvents.length) {
   389	      return null;
   390	    }
   391	
   392	    // NoteEvent.start/duration are in BEATS, but Tone.js interprets numeric time and
   393	    // duration values as SECONDS (Transport.bpm only affects tick/notation times). Convert
   394	    // beats -> seconds once here so the tempo actually applies and playback length matches
   395	    // the auto-stop math in play() (which is already computed in seconds).
   396	    const secondsPerBeat = 60 / tempo;
   397	    const events = noteEvents.map((event) => ({
   398	      time: event.start * secondsPerBeat,
   399	      note: event.note,
   400	      duration: event.duration * secondsPerBeat,
   401	      velocity: event.velocity,
   402	      pan: event.pan || 0,
   403	      reverbSend: event.effects?.reverbSend || 0.2,
   404	      // Left undefined when absent so the filter keeps its static preset frequency.
   405	      filterCutoff: event.effects?.filterCutoff,
   406	      voice: event.trackId as VoiceType | undefined,
   407	    }));
   408	
   409	    const tone = toneRef.current;
   410	    if (!tone) {
   411	      return null;
   412	    }
   413	
   414	    const part = new tone.Part((time, value) => {
   415	      if (isMultiVoice && value.voice) {
   416	        const voiceSynth = voiceSynthsRef.current.get(value.voice);
   417	        if (!voiceSynth) return;
   418	
   419	        voiceSynth.reverb.wet.setValueAtTime(value.reverbSend, time);
   420	        voiceSynth.panner.pan.setValueAtTime(value.pan, time);
   421	
   422	        if (value.filterCutoff != null) {
   423	          voiceSynth.filter.frequency.setValueAtTime(
   424	            filterCutoffToFrequency(value.filterCutoff),
   425	            time
   426	          );
   427	        }
   428	
   429	        voiceSynth.synth.triggerAttackRelease(value.note, value.duration, time, value.velocity);
   430	        return;
   431	      }
   432	
   433	      if (!synthRef.current || !reverbRef.current || !pannerRef.current) {
   434	        return;
   435	      }
   436	
   437	      pannerRef.current.pan.setValueAtTime(value.pan, time);
   438	      reverbRef.current.wet.setValueAtTime(value.reverbSend, time);
   439	
   440	      if (value.filterCutoff != null && filterRef.current) {
   441	        filterRef.current.frequency.setValueAtTime(
   442	          filterCutoffToFrequency(value.filterCutoff),
   443	          time
   444	        );
   445	      }
   446	
   447	      synthRef.current.triggerAttackRelease(value.note, value.duration, time, value.velocity);
   448	    }, events);
   449	
   450	    part.loop = false;
   451	    partRef.current = part;
   452	    tone.Transport.bpm.value = tempo;
   453	
   454	    return part;
   455	  }, [disposePart, isMultiVoice, noteEvents, tempo]);
   456	
   457	  useEffect(() => {
   458	    if (toneRef.current) {
   459	      toneRef.current.Transport.bpm.value = tempo;
   460	    }
   461	  }, [tempo]);
   462	
   463	  // Update current time while playing
   464	  useEffect(() => {
   465	    if (!isPlaying) return;
   466	
   467	    let frameId: number;
   468	
   469	    const updateTime = () => {
   470	      if (toneRef.current) {
   471	        setCurrentTime(toneRef.current.Transport.seconds);
   472	      }
   473	      frameId = requestAnimationFrame(updateTime);
   474	    };
   475	
   476	    frameId = requestAnimationFrame(updateTime);
   477	
   478	    return () => cancelAnimationFrame(frameId);
   479	  }, [isPlaying]);
   480	
   481	  const play = useCallback(async () => {
   482	    if (!noteEvents.length) {
   483	      console.warn('No note events to play');
   484	      return;
   485	    }
   486	
   487	    try {
   488	      await ensureAudioGraph();
   489	      const part = buildPart();
   490	
   491	      if (!part) {
   492	        console.warn('No part to play');
   493	        return;
   494	      }
   495	
   496	      if (stopSchedulerIdRef.current !== null) {
   497	        toneRef.current?.Transport.clear(stopSchedulerIdRef.current);
   498	        stopSchedulerIdRef.current = null;
   499	      }
   500	
   501	      if (!toneRef.current) {
   502	        return;
   503	      }
   504	
   505	      toneRef.current.Transport.stop();
   506	      toneRef.current.Transport.position = 0;
   507	      setCurrentTime(0);
   508	
   509	      part.start(0);
   510	      toneRef.current.Transport.start();
   511	      setIsPlaying(true);
   512	
   513	      const durationBeats = noteEvents.reduce(
   514	        (max, event) => Math.max(max, event.start + event.duration),
   515	        0
   516	      );
   517	      const durationSeconds = (durationBeats + 1) * (60 / tempo);
   518	
   519	      stopSchedulerIdRef.current = toneRef.current.Transport.scheduleOnce(() => {
   520	        stop();
   521	      }, `+${durationSeconds}`);
   522	    } catch (error) {
   523	      console.error('Failed to play composition:', error);
   524	      stop();
   525	    }
   526	  }, [buildPart, ensureAudioGraph, noteEvents, stop, tempo]);
   527	
   528	  // Cleanup on unmount
   529	  useEffect(() => {
   530	    return () => {
   531	      stop();
   532	      disposeAudioGraph();
   533	    };
   534	  }, [disposeAudioGraph, stop]);
   535	
   536	  return {
   537	    play,
   538	    stop,
   539	    isPlaying,
   540	    isLoading,
   541	    currentTime,
   542	  };
   543	}
