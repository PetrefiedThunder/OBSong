   141	export function noteNameToMidi(noteName: string): number {
   142	  // Allow a negative octave (e.g. "C-1", which midiToNoteName emits for MIDI 0-11).
   143	  const match = noteName.match(/^([A-G]#?)(-?\d+)$/);
   144	  if (!match) {
   145	    throw new Error(`Invalid note name: ${noteName}`);
   146	  }
   147	
   148	  const [, note, octaveStr] = match;
   149	  const noteIndex = CHROMATIC_NOTES.indexOf(note);
   150	  // The regex allows e.g. "E#"/"B#", which aren't in CHROMATIC_NOTES. Reject them instead
   151	  // of silently returning a wrong MIDI number (indexOf would be -1).
   152	  if (noteIndex === -1) {

   154	  }
